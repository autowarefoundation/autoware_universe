cmake_minimum_required(VERSION 3.16)
project(camp_installed_consumer LANGUAGES CXX)
find_package(autoware_camp_selector CONFIG REQUIRED)
add_executable(camp_installed_check main.cpp)
target_link_libraries(camp_installed_check PRIVATE autoware::camp_selector)
enable_testing()
if(NOT EXISTS "${CAMP_MODEL_PATH}")
  message(FATAL_ERROR "Pass CAMP_MODEL_PATH from the installed package's share directory")
endif()
add_test(NAME installed_nonzero_selection COMMAND camp_installed_check "${CAMP_MODEL_PATH}")
